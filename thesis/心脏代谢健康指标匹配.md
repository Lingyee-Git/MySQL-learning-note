
## 1. 就诊事件大表

```sql
CREATE TABLE `就诊事件大表` AS

SELECT `身份证号`, '门诊' `事件类型` , `就诊ID号` `事件标识号`
FROM `门诊信息-pcos筛选（未排除？）`

UNION ALL

SELECT `身份证号`, '住院' `事件类型` , `住院号` `事件标识号`
FROM `住院信息-pcos筛选`
```

## 2. 就诊事件-检查指标全表

```sql
CREATE TABLE `就诊事件匹配检查结果` AS

WITH 
m1 AS (
SELECT
  A.`事件标识号`,
  A.`身份证号`,
  '门诊化验' AS `数据来源`,
  B.`检验项目名称` AS `项目名称`,
  B.`子项名称`,
  B.`结果正常标志` AS `结果`,
  B.`报告时间`
FROM `就诊事件大表` A
LEFT JOIN `门诊化验` B
ON A.`事件标识号` = B.`就诊ID号`
WHERE A.`事件类型` = '门诊'
),

m2 AS (
SELECT
  A.`事件标识号`,
  A.`身份证号`,
  '门诊检查' AS `数据来源`,
  B.`检查项目名称` AS `项目名称`,
  NULL AS `子项名称`,
  B.`检查结果` AS `结果`,
  B.`报告日期` AS `报告时间`
FROM `就诊事件大表` A
LEFT JOIN `门诊检查` B
ON A.`事件标识号` = B.`就诊ID号`
WHERE A.`事件类型` = '门诊'
),

z1 AS (
SELECT
  A.`事件标识号`,
  A.`身份证号`,
  '住院化验' AS `数据来源`,
  B.`检验项目名称` AS `项目名称`,
  B.`子项名称`,
  B.`结果正常标志` AS `结果`,
  B.`报告时间`
FROM `就诊事件大表` A
LEFT JOIN `住院化验` B
ON A.`事件标识号` = B.`住院号`
WHERE A.`事件类型` = '住院'
),

z2 AS (
SELECT
  A.`事件标识号`,
  A.`身份证号`,
  '住院检查' AS `数据来源`,
  B.`检查项目名称` AS `项目名称`,
  NULL AS `子项名称`,
  B.`检查结果` AS `结果`,
  B.`报告日期` AS `报告时间`
FROM `就诊事件大表` A
LEFT JOIN `住院检查` B
ON A.`事件标识号` = B.`住院号`
WHERE A.`事件类型` = '住院'
)

SELECT * FROM m1
UNION ALL
SELECT * FROM m2
UNION ALL
SELECT * FROM z1
UNION ALL
SELECT * FROM z2
```
